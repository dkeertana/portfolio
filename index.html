public class RFQComplianceEngine {
  public ComplianceReport evaluate(RFQSpec target, VendorBid bid) {
    Matrix matchScore = new RuleEngine()
      .verifyTolerances(target.getConstraints())
      .validateCertifications(bid.getCerts())
      .computeRiskIndex();
    
    return new ComplianceReport(
      matchScore.isSatisfied(),
      matchScore.getVarianceDeltas()
    );
  }
}
